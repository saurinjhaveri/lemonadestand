const rnd = (a,b) => Math.floor(Math.random()*(b-a+1))+a;

const clamp = (v,a,b) => Math.max(a,Math.min(b,v));

const fmt$ = v => (v<0?'-':'')+'$'+Math.abs(v).toFixed(2);


function quality(r) {

  const dl=Math.abs(r.lemons-5), ds=Math.abs(r.sugar-5), di=Math.abs(r.ice-5);

  const dev = dl+ds+di;

  const q = clamp(1-dev/15, 0, 1);

  const gross = (r.lemons>=9&&r.sugar<=2)||(r.sugar>=9&&r.lemons<=2)||

                (r.ice>=9&&r.lemons<=2)||(r.lemons<=1&&r.sugar<=1&&r.ice<=1);

  return {q, gross};

}

function batchCost(r) { return r.lemons*ICOST.lemons + r.sugar*ICOST.sugar + r.ice*ICOST.ice; }


function pickWeather() {

  if(r<0.60) return 'OVER';

  if(r<0.75) return 'RAINY';

  if(r<0.85) return 'STORMY';

  return 'WINDY';

}

function pickEvent() {

  if(Math.random()>0.15) return null;

// ── CUSTOMER SIM ──────────────────────────────────────────

function buildCustomers() {

  const w = WEATHER[G.weather];

  const {q, gross} = quality(G.recipe);

  let base = Math.round(w.d * rnd(8,16));

  if(G.event==='SCHOOL') base = Math.round(base*2);

  if(G.event==='HEATWAVE'&&G.weather==='SUNNY') base = Math.round(base*1.5);

  base = Math.max(0, base);


  const skins=[PAL.SKIN,PAL.SKIN2,PAL.SKIN3,'#C68642','#8D5524'];

  const shirts=['#E53935','#8E24AA','#1E88E5','#00897B','#F4511E','#6D4C41','#3949AB','#00ACC1'];


  return Array.from({length:base}, (_,i) => {

    let em, cups=0, tip=0, buys=true;

    if(gross){

      em=EM.GROSS; buys=false;

    } else if(G.weather==='SUNNY'&&q>0.5&&Math.random()<0.4){

      em=EM.HOT; cups=rnd(2,4);

    } else if((G.weather==='RAINY'||G.weather==='STORMY')&&Math.random()<0.35){

      em=EM.COLD; buys=Math.random()<0.4; cups=buys?1:0;

    } else if(G.price<=0.75&&q>=0.7){

      em=EM.LOVE; cups=rnd(2,4); tip=Math.random()<0.5?0.25:0;

    } else if(G.price<=1.5&&q>=0.45){

      em=EM.HAPPY; cups=rnd(1,2);

    } else if(G.price>3.0){

      em=EM.ANNOY; buys=false;

    } else if(q<0.3){

      em=EM.MEH; buys=Math.random()<0.5; cups=1;

    } else {

      em=EM.HAPPY; cups=1;

    }

    if(!buys) cups=0;

    return {

      id:i, em, cups, tip,

      skin: skins[rnd(0,4)], shirt: shirts[rnd(0,7)],

      hair:['#1a1a1a','#8B4513','#FFD700','#C0392B','#2C3E50'][rnd(0,4)],

      umbrella: G.weather==='RAINY'||G.weather==='STORMY',

      x:-35, y:285, tx: 130+rnd(-10,10), state:'walk', etimer:0, flip:false,

    };

  });

}

function lighten(hex){

  const r=parseInt(hex.slice(1,3),16),g=parseInt(hex.slice(3,5),16),b=parseInt(hex.slice(5,7),16);

  const mix=(v,t=40)=>Math.min(255,v+t);

  return `rgb(${mix(r)},${mix(g)},${mix(b)})`;

}

function pixelSprite(px,x,y,scale,colors){

function drawMiniGraph(x,y,w,h){

// ── UPDATE / GAME LOGIC ───────────────────────────────────

let dayEnded=false;

function update(){

  G.frame++;

  if(G.state===S.BUSY){

    G.custTimer++;

    if(G.custTimer>=G.custInterval && G.cIdx<G.customers.length){

      const cdata=G.customers[G.cIdx];

      G.customers_active.push({...cdata, x:-35,y:285,tx:120+rnd(-15,15),state:'walk',etimer:0,flip:false});

      if(cdata.cups>0){

        G.cupsToday+=cdata.cups;

        G.revToday+=cdata.cups*G.price+cdata.tip;

      }

      G.cIdx++;

      G.custTimer=0;

    }

    for(let i=G.customers_active.length-1;i>=0;i--){

      const c=G.customers_active[i];

      if(c.state==='walk'){

        c.x+=2;

        if(c.x>=c.tx){c.state='react';c.etimer=55;}

      } else if(c.state==='react'){

        c.etimer--;

        if(c.etimer<=0){c.state='leave';c.flip=true;}

      } else if(c.state==='leave'){

        c.x-=2;

        if(c.x<-40) G.customers_active.splice(i,1);

      }

    }

    if(!dayEnded && G.cIdx>=G.customers.length && G.customers_active.length===0){

      dayEnded=true;

      setTimeout(()=>{endDay();dayEnded=false;},400);

    }

  }

}


function startDay(){

  G.weather=pickWeather();

  G.event=pickEvent();

  G.customers=buildCustomers();

  G.cIdx=0; G.customers_active=[];

  G.cupsToday=0; G.revToday=0; G.costToday=0; G.profToday=0;

  G.custTimer=0; G.custInterval=55; G.weatherTimer=0;

  G.rainDrops=[]; G.rain=[];

  G.state=S.WEATHER;

  dayEnded=false;

}

function openBusiness(){G.state=S.BUSY;}

let _endDayCalled=false;

function endDay(){

  if(_endDayCalled) return;

  if(G.state!==S.BUSY&&G.state!==S.FF) return;

  _endDayCalled=true;

  setTimeout(()=>{_endDayCalled=false;},200);

  const batches=Math.max(1,Math.ceil(G.cupsToday/10));

  G.costToday=batches*batchCost(G.recipe);

  if(G.event==='HEALTH'&&quality(G.recipe).q<0.4) G.costToday+=2.00;

  G.profToday=G.revToday-G.costToday;

  G.money+=G.profToday;

  const prevCum=G.history.length?G.history[G.history.length-1].cumulativeProfit:0;

  G.history.push({day:G.day,cups:G.cupsToday,revenue:G.revToday,cost:G.costToday,profit:G.profToday,cumulativeProfit:prevCum+G.profToday,weather:G.weather,event:G.event});

  G.customers_active=[];

  if(G.money<0){G.state=S.BANKRUPT;return;}

  G.state=S.SUMMARY;

}

function advanceDay(){

  if(G.day>=30){G.state=S.GAMEOVER;return;}

  G.day++;G.state=S.SETUP;

}


function fastForwardDay(){

  for(let i=G.cIdx;i<G.customers.length;i++){

    const c=G.customers[i];

    if(c.cups>0){G.cupsToday+=c.cups;G.revToday+=c.cups*G.price+c.tip;}

  }

  G.cIdx=G.customers.length; G.customers_active=[];

  dayEnded=false; _endDayCalled=false;

  endDay();

}


function fastForwardAll(){

  if(G.ffRunning) return;

  G.ffRunning=true;

  for(let i=G.cIdx;i<G.customers.length;i++){

    const c=G.customers[i];

    if(c.cups>0){G.cupsToday+=c.cups;G.revToday+=c.cups*G.price+c.tip;}

  }

  G.cIdx=G.customers.length; G.customers_active=[];

  _endDayCalled=false;

  G.state=S.FF;

  endDay();

  if(G.state===S.BANKRUPT){G.ffRunning=false;return;}

  const runNext=()=>{

    if(G.day>=30||G.state===S.BANKRUPT){

      G.ffRunning=false;

      if(G.state!==S.BANKRUPT) G.state=S.GAMEOVER;

      return;

    }

    G.day++;G.state=S.FF;

    G.weather=pickWeather();G.event=pickEvent();

    G.customers=buildCustomers();

    let tc=0,tr=0;

    for(const c of G.customers){if(c.cups>0){tc+=c.cups;tr+=c.cups*G.price+c.tip;}}

    G.cupsToday=tc;G.revToday=tr;

    const batches=Math.max(1,Math.ceil(tc/10));

    G.costToday=batches*batchCost(G.recipe);

    if(G.event==='HEALTH'&&quality(G.recipe).q<0.4) G.costToday+=2;

    G.profToday=G.revToday-G.costToday;

    G.money+=G.profToday;

    const prevCum=G.history.length?G.history[G.history.length-1].cumulativeProfit:0;

    G.history.push({day:G.day,cups:tc,revenue:tr,cost:G.costToday,profit:G.profToday,cumulativeProfit:prevCum+G.profToday,weather:G.weather,event:G.event});

    G.ffFlash={day:G.day,weather:G.weather,profit:G.profToday,cups:tc,t:22};

    if(G.money<0){G.ffRunning=false;G.state=S.BANKRUPT;return;}

    setTimeout(runNext,320);

  };

  setTimeout(runNext,10);

}

function applyDrag(mx){

  const d=G.mouse.drag;

  if(!d) return;

  if(d.key==='price'){

    const pct=clamp((mx-516)/190,0,1);

    G.price=Math.round((0.25+pct*4.75)/0.25)*0.25;

    G.price=clamp(G.price,0.25,5.00);

  } else {

    const pct=clamp((mx-516)/190,0,1);

    G.recipe[d.key]=Math.max(1,Math.min(10,Math.round(pct*9+1)));

  }

}


function handleClick(mx,my){

  if(G.state===S.SETUP){

    for(const hk of Object.keys(G.hz)){

      if(hk.startsWith('sl_')){

        const h=G.hz[hk];

        if(mx>=h.x&&mx<=h.x+h.w&&my>=h.y&&my<=h.y+h.h){

          G.mouse.drag={key:h.key};

          applyDrag(mx);

          return;

        }

      }

    }

  }

  for(const [id,h] of Object.entries(G.hz)){

    if(id.startsWith('sl_')) continue;

    if(mx>=h.x&&mx<=h.x+h.w&&my>=h.y&&my<=h.y+h.h){

      handleAction(id);

      return;

    }

  }

}


function handleAction(id){

  switch(id){

    case 'startGame':

      G.day=1;G.money=20;G.history=[];G.recipe={lemons:5,sugar:5,ice:5};G.price=1.00;

      G.state=S.SETUP;break;

    case 'start': startDay();break;

    case 'openBiz': openBusiness();break;

    case 'ff1': fastForwardDay();break;

    case 'ff2': fastForwardAll();break;

    case 'next': advanceDay();break;

    case 'restart': initGame();break;

  }

}


document.addEventListener('keydown',e=>{

  if(e.key==='Enter'||e.key===' '){

    e.preventDefault();

    if(G.state===S.START) handleAction('startGame');

    else if(G.state===S.WEATHER) handleAction('openBiz');

    else if(G.state===S.SUMMARY) handleAction('next');

    else if(G.state===S.GAMEOVER||G.state===S.BANKRUPT) handleAction('restart');

  }

});
